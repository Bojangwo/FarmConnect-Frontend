  const [products, setProducts] = useState([]);

        const [productsData, categoriesData] = await Promise.all([
          getMarketplaceProducts(),
          getCategories(),

        if (!active) return;

        const productList = Array.isArray(productsData)
          ? productsData
          : productsData.products || [];

        setProducts(productList);

        setCategories(
          Array.isArray(categoriesData)
            ? categoriesData
            : categoriesData.categories || []
        );

        const farmerIds = [
          ...new Set(
            productList
              .map((product) => product.farmer?._id)
              .filter(Boolean)
          ),
        ];

        const results = await Promise.allSettled(
          farmerIds.map(getFarmerReviews)
        );

        if (!active) return;

        const ratings = {};

        results.forEach((result, index) => {
          if (result.status === "fulfilled") {
            ratings[farmerIds[index]] = {
              averageRating: result.value.averageRating,
              totalReviews: result.value.totalReviews,
            };
          }
        });

        setFarmerRatings(ratings);
      } catch (err) {
        if (active) {
          setError(err.message || "Unable to load marketplace.");
        }

        if (active) setLoading(false);
      }

    return () => {
      active = false;

  const filteredProducts = products

      const matchesSearch =
        !term ||
        product.name?.toLowerCase().includes(term) ||
        product.farmer?.name?.toLowerCase().includes(term);

      const categoryId =
        typeof product.category === "object"
          ? product.category?._id
          : product.category;

      return (
        matchesSearch &&
        (!selectedCategory || categoryId === selectedCategory)
      );
    })
    .sort((a, b) => {
      if (sortBy === "price-low") {
        return Number(a.price) - Number(b.price);

      if (sortBy === "price-high") {
        return Number(b.price) - Number(a.price);
      }

      if (sortBy === "name") {
        return a.name.localeCompare(b.name);
      }

      return new Date(b.createdAt || 0) - new Date(a.createdAt || 0);
    });

    new Intl.NumberFormat("en-GM", {
      maximumFractionDigits: 2,

    if (!image) return null;

    const imagePath =
      typeof image === "string"
        ? image
        : image.url || image.path;

    if (!imagePath) return null;

    if (/^https?:\/\//i.test(imagePath)) {
      return imagePath;

    // Set this to your backend URL if your API uses

      "http://localhost:5000";

    return `${backendUrl.replace(/\/$/, "")}/${imagePath.replace(
      /^\/+/,
      ""

  };

  return (

                Straight to you.

            </h1>

            <p className="mt-6 max-w-lg text-base leading-8 text-white/75">
              Discover quality farm produce, connect directly with

              FarmConnect.

            <a

            >
              Explore Products
              <span aria-hidden="true">↗</span>
            </a>
          </div>

          <div className="relative hidden md:block">
            <div className="rounded-[2rem] border border-white/15 bg-white/10 p-5 shadow-2xl backdrop-blur-sm">
              <div className="flex min-h-72 flex-col items-center justify-center rounded-[1.5rem] border border-dashed border-white/20 bg-[#24543D] p-8 text-center">
                <span className="text-7xl">🥬</span>
                <p className="mt-5 text-2xl font-bold">
                  Naturally fresh.
                </p>
                <p className="mt-2 text-sm text-white/70">
                  Local produce. Real farmers. Better connections.
                </p>
                <div className="mt-7 flex gap-3 text-4xl">
                  🍅 🥕 🌽 🍆
                </div>
              </div>
            </div>
